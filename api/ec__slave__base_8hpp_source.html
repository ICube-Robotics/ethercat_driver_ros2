47constexpr uint8_t kAlStateInit = 1;

52inline std::string alStateStr(uint8_t al_state)
53{
54 switch (al_state) {
55 case kAlStateInit: return "INIT";
56 case kAlStatePreop: return "PREOP";
57 case kAlStateSafeop: return "SAFEOP";
58 case kAlStateOp: return "OP";
59 default: return "UNKNOWN";
60 }
61}

153 virtual void collectDiagnostics(diagnostic_msgs::msg::DiagnosticStatus & status) const
154 {
155 status.level = (online_ && al_state_ == kAlStateOp) ?
156 diagnostic_msgs::msg::DiagnosticStatus::OK :
157 diagnostic_msgs::msg::DiagnosticStatus::ERROR;
158 status.message = online_ ? "online" : "offline";
159 diagnostic_msgs::msg::KeyValue online_kv;
160 online_kv.key = "online";
161 online_kv.value = online_ ? "true" : "false";
162 status.values.push_back(online_kv);
163 diagnostic_msgs::msg::KeyValue al_state_kv;
164 al_state_kv.key = "al_state";
165 al_state_kv.value = alStateStr(al_state_);
166 status.values.push_back(al_state_kv);
167 if (!process_data_fault_.empty()) {
168 status.level = diagnostic_msgs::msg::DiagnosticStatus::ERROR;
169 status.message = "process_data fault: " + process_data_fault_;
170 diagnostic_msgs::msg::KeyValue fault_kv;
171 fault_kv.key = "process_data_fault";
172 fault_kv.value = process_data_fault_;
173 status.values.push_back(fault_kv);
