162 return false;
163 }
164 // optional human-readable label, surfaced in the failure message
165 if (sdo_config["description"]) {
166 description = sdo_config["description"].as<std::string>();
167 }
168 // allowed value(s): either a single scalar `value`, or a list `values`
169 allowed_values.clear();
170 if (sdo_config["values"]) {
171 for (const auto & v : sdo_config["values"]) {
172 allowed_values.push_back(v.as<int64_t>());
173 }
174 } else if (sdo_config["value"]) {
175 allowed_values.push_back(sdo_config["value"].as<int64_t>());
176 } else {
177 std::cerr << "sdo_check " << index << ": missing 'value' or 'values'" << std::endl;
178 return false;
179 }
180 if (allowed_values.empty()) {
181 std::cerr << "sdo_check " << index << ": 'values' is empty" << std::endl;
182 return false;
183 }
184 return true;
185 }

202 if (data_type == "uint64") {return static_cast<int64_t>(read_u64(buffer));}
203 if (data_type == "int64") {return read_s64(buffer);}
204 return 0;
205 }
